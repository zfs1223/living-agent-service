public enum RequirementStatus {
    /** 初始状态：用户首次表达需求，尚未确认 */    
DRAFT,    
    /** 主脑判断需求缺关键要素，必须追问 */    
NEEDS_CLARIFICATION,    
    /** 已发出澄清问题，等待用户回答 */    
CLARIFICATION_PENDING,    
    /** 需求已明确，可进入正式规划与分派 */    
REQUIREMENT_CONFIRMED,    
    /** 主脑生成正式任务计划中 */    
PLANNING,    
    /** 计划已生成，等待分派 */    
PLANNED,    
    /** 员工任务单已生成并分派 */    
ASSIGNED,    
    /** 员工正在执行 */    
EXECUTING,    
    /** 验收通过并完成 */    
COMPLETED,    
    /** 执行失败或验收失败 */    
FAILED;    
    /**
     * 状态转移规则
     */    
public static boolean canTransition(RequirementStatus from, RequirementStatus to) {        
return switch (from) {            
case DRAFT -> to == NEEDS_CLARIFICATION || to == REQUIREMENT_CONFIRMED;            
case NEEDS_CLARIFICATION -> to == CLARIFICATION_PENDING;            
case CLARIFICATION_PENDING -> to == REQUIREMENT_CONFIRMED || to == NEEDS_CLARIFICATION;            
case REQUIREMENT_CONFIRMED -> to == PLANNING;            
case PLANNING -> to == PLANNED;            
case PLANNED -> to == ASSIGNED;            
case ASSIGNED -> to == EXECUTING;            
case EXECUTING -> to == COMPLETED || to == FAILED;            
case COMPLETED, FAILED -> false; // 终态，不可转移        
};    
}    
    /**
     * 检查是否允许进入分派阶段
     */    
public boolean allowsAssignment() {        
return this == PLANNED;    
}    
    /**
     * 检查是否允许生成员工执行产物
     */    
public boolean allowsExecution() {        
return this == ASSIGNED || this == EXECUTING;    
}    
    /**
     * 检查是否需要澄清
     */    
public boolean needsClarification() {        
return this == DRAFT || this == NEEDS_CLARIFICATION || this == CLARIFICATION_PENDING;    
}
}

public class DefaultRequirementReadinessEvaluator implements RequirementReadinessEvaluator {        
// ========== 评估维度权重 ==========    
private static final double WEIGHT_ACTION = 0.25;    // 动作明确性    
private static final double WEIGHT_TARGET = 0.25;    // 目标产物    
private static final double WEIGHT_CONTEXT = 0.15;    // 上下文    
private static final double WEIGHT_SCOPE = 0.15;     // 范围边界    
private static final double WEIGHT_CRITERIA = 0.10;  // 验收标准    
private static final double WEIGHT_COMPLETENESS = 0.10; // 完整性        
// ========== 判定阈值 ==========    
private static final double THRESHOLD_READY = 0.85;    
private static final double THRESHOLD_PARTIALLY_READY = 0.65;        
// ========== 关键词定义 ==========    
private static final List<String> ACTION_KEYWORDS = List.of(        
"开发", "生成", "创建", "制作", "编写", "设计", "分析", "修改", "修复",        
"构建", "部署", "实现", "搭建", "写", "做", "画", "算",        
"build", "create", "generate", "design", "develop", "fix", "implement"    
);        
private static final List<String> TARGET_KEYWORDS = List.of(        
"网页", "页面", "网站", "文档", "报告", "方案", "代码", "接口", "系统",        
"游戏", "小程序", "app", "h5", "表格", "图表", "流程",        
"web", "page", "document", "report", "code", "api", "system"    
);        
private static final List<String> SCOPE_KEYWORDS = List.of(        
"范围", "包括", "包含", "只要", "限定", "只做",        
"scope", "include", "only", "limit"    
);        
private static final List<String> CRITERIA_KEYWORDS = List.of(        
"标准", "要求", "必须", "满足", "验证", "通过",        
"criteria", "requirement", "must", "pass", "verify"    
);        
@Override    
public RequirementReadinessResult evaluate(            
String userMessage,             
String department,             
String sessionId) {                
if (userMessage == null || userMessage.isBlank()) {            
return RequirementReadinessResult.insufficient(                
List.of("请描述您需要完成的任务"),                
"Empty message"            
);        
}                
String lower = userMessage.toLowerCase(Locale.ROOT);                
// 1. 动作明确性评估        
boolean hasAction = containsAny(lower, ACTION_KEYWORDS);        
double actionScore = hasAction ? WEIGHT_ACTION : 0;                
// 2. 目标产物评估        
boolean hasTarget = containsAny(lower, TARGET_KEYWORDS);        
double targetScore = hasTarget ? WEIGHT_TARGET : 0;                
// 3. 上下文充分性评估        
boolean hasContext = lower.length() >= 20;        
double contextScore = hasContext ? WEIGHT_CONTEXT : 0;                
// 4. 范围明确性评估        
boolean hasScope = containsAny(lower, SCOPE_KEYWORDS);        
double scopeScore = hasScope ? WEIGHT_SCOPE : 0;                
// 5. 验收标准隐含评估        
boolean hasCriteria = containsAny(lower, CRITERIA_KEYWORDS);        
double criteriaScore = hasCriteria ? WEIGHT_CRITERIA : WEIGHT_CRITERIA * 0.5;                
// 6. 完整性评估        
double completenessScore = Math.min(WEIGHT_COMPLETENESS,             
(lower.length() / 200.0) * WEIGHT_COMPLETENESS);                
// 计算总置信度        
double confidence = actionScore + targetScore + contextScore + 
                           scopeScore + criteriaScore + completenessScore;                
// 收集缺失元素        
List<String> missing = new ArrayList<>();        
if (!hasAction) missing.add("明确的动作/操作");        
if (!hasTarget) missing.add("目标产物类型");        
if (!hasContext) missing.add("充分的上下文信息");        
if (!hasScope) missing.add("明确的任务范围边界");        
if (!hasCriteria && confidence < THRESHOLD_READY)             
missing.add("验收标准或要求");                
// 生成澄清问题        
List<String> questions = generateClarificationQuestions(missing, hasAction, hasTarget);                
// 根据阈值返回结果        
if (confidence >= THRESHOLD_READY && missing.isEmpty()) {            
return RequirementReadinessResult.sufficient(confidence,                
String.format("READY: Action=%s, Target=%s, Context=%s", hasAction, hasTarget, hasContext));        
}                
if (confidence >= THRESHOLD_PARTIALLY_READY) {            
return RequirementReadinessResult.partiallySufficient(confidence,
                missing, questions,                
String.format("PARTIALLY_READY: missing=%s", missing));        
}                
return RequirementReadinessResult.insufficient(questions,            
String.format("NEEDS_CLARIFICATION: confidence=%.2f, missing=%s", confidence, missing));    
}        
private List<String> generateClarificationQuestions(            
List<String> missing, boolean hasAction, boolean hasTarget) {                
List<String> questions = new ArrayList<>();                
if (!hasAction) {            
questions.add("请问您希望执行什么操作？（如：开发、生成、分析、修改等）");        
}        
if (!hasTarget) {            
questions.add("请明确您期望的产出物类型？（如：网页、文档、代码、数据分析报告等）");        
}        
if (missing.contains("明确的任务范围边界")) {            
questions.add("请说明任务的边界和范围（如：只需要XX功能，不需要YY）");        
}        
if (missing.contains("验收标准或要求")) {            
questions.add("您对交付结果有什么具体的验收标准或要求吗？");        
}                
// 限制最多3个问题        
return questions.stream().limit(3).toList();    
}        
private boolean containsAny(String text, List<String> keywords) {        
return keywords.stream().anyMatch(text::contains);    
}
}

// 使用示例
public class RequirementStatusDemo {        
public static void main(String[] args) {        
// 正常流程        
assert RequirementStatus.DRAFT.canTransition(RequirementStatus.NEEDS_CLARIFICATION);        
assert RequirementStatus.NEEDS_CLARIFICATION.canTransition(RequirementStatus.CLARIFICATION_PENDING);        
assert RequirementStatus.CLARIFICATION_PENDING.canTransition(RequirementStatus.REQUIREMENT_CONFIRMED);        
assert RequirementStatus.REQUIREMENT_CONFIRMED.canTransition(RequirementStatus.PLANNING);        
assert RequirementStatus.PLANNING.canTransition(RequirementStatus.PLANNED);        
assert RequirementStatus.PLANNED.canTransition(RequirementStatus.ASSIGNED);        
assert RequirementStatus.ASSIGNED.canTransition(RequirementStatus.EXECUTING);        
assert RequirementStatus.EXECUTING.canTransition(RequirementStatus.COMPLETED);                
// 非法转移（应该抛出异常或返回false）        
assert !RequirementStatus.DRAFT.canTransition(RequirementStatus.ASSIGNED); // DRAFT不能直接到ASSIGNED        
assert !RequirementStatus.NEEDS_CLARIFICATION.canTransition(RequirementStatus.PLANNING); // 未确认不能规划        
assert !RequirementStatus.COMPLETED.canTransition(RequirementStatus.EXECUTING); // 终态不可转移                
// 状态检查        
assert RequirementStatus.PLANNED.allowsAssignment();        
assert RequirementStatus.ASSIGNED.allowsExecution();        
assert RequirementStatus.DRAFT.needsClarification();        
assert !RequirementStatus.PLANNING.needsClarification();    
}
}
